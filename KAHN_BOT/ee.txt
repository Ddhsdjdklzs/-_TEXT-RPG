function nuum(num){
  
  if(num == 0) return 0;
  
  return (String(num).split('').reverse().reduce((acc, curr, index) => acc.concat((index + 1) % 4 === 0 ? [curr, (['만', '억', '조', '경', '해', '자', '양', '구', '간', '정', '재', '극', '항하사', '아승기', '나유타', '불가사의', '무량대수'][index / 4 | 0] || '').concat(' ')] : curr), []).reverse().join('').split(/(\D+)(\d{4})/).map((e) => /\D/.test(e) ? e : (Number(e) || '')).join('').replace(/^\D+| \D+/g, ' ').trim());
  
}

function gauge(num, max, p, n){
  
    let bar = ['▏', '▏', '▎', '▍', '▌', '▋', '▊', '▉'];
    
    let per = 100/(max/num)/10;
    
    let gauge = [];
    
    for(let i=0; i<parseInt(per); i++) gauge.push('█');
    
    if(per != parseInt(per)) gauge.push(bar[parseInt((per-gauge.length)*10/1.25)]);
        
    for(let i=gauge.length; i<10; i++) gauge.push(' ');
    
    if(!p) p = 0; if(!n) n = 0;
    
    return gauge.join('')+
    
        (!p?'': ' ('+(per*10).toFixed(0) + '%)')+
        
        (!n?'':' ('+num+'/'+max+')');
        
}

function makeMaze(yyy,xxx){

  yyy |= 1;

  xxx |= 1;

  var arr = Array(yyy).fill().map(a => Array(xxx).fill('■'));

  var visit = Array(yyy).fill().map(a => Array(xxx).fill(false));

  var pq = new java.util.PriorityQueue;

  var tmp = [];

  tmp[1] = [0, 0];

  pq.add(1);

  arr[0][0] = '□';

  visit[0][0] = true;

  var dir = [1, 0, 0, 1, -1, 0, 0, -1];

  while(!pq.empty){

    var r = Number(pq.poll());

    var y = tmp[r][0];
    
    var x = tmp[r][1];

    delete tmp[r];

    for(var i = 0; i < 4; i++){

      var yy = y + dir[i * 2];

      var xx = x + dir[i * 2 + 1];

      if(yy >= 0 && xx >= 0 && yy < yyy / 2 && xx < xxx / 2 && !visit[yy][xx]){

      visit[yy][xx] = true;

      arr[yy * 2][xx * 2] = '□';
      
      arr[yy * 2 - dir[i * 2]][xx * 2 - dir[i * 2 + 1]] = '□';

      var rn = Math.random();

      while(rn in tmp) rn = Math.random();

      tmp[rn] = [yy, xx];

      pq.add(rn);

      }

    }

  }

  delete pq;

  delete visit;

  delete tmp;

  return arr.map(a => a.join('')).join('\n');

}
